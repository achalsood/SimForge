# Tool executables retain full Debug symbols, but do not need incremental-link databases.
if(MSVC)
    get_property(tool_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(tool_target IN LISTS tool_targets)
        get_target_property(tool_type ${tool_target} TYPE)
        if(tool_type STREQUAL "EXECUTABLE")
            target_link_options(${tool_target} PRIVATE "$<$<CONFIG:Debug>:/INCREMENTAL:NO>")
        endif()
    endforeach()
endif()
